Private Sub localiza_AfterUpdate()

    ' Encontrar o registro que coincide com o controle.
    Dim rs As Object

    Set rs = Me.Recordset.Clone
    rs.FindFirst "[cod_tipo_servico] = " & Str(Me![localiza])
    Me.Bookmark = rs.Bookmark

End Sub


Private Sub localiza1_Change()

    On Error Resume Next
    Dim I As Integer
    Dim strProcura As String, intPos As Integer
    Dim strList As String
    strProcura = UCase$(Me.localiza1.Text)
    intPos = Len(strProcura)
    If intPos Then
        For I = 0 To localiza.ListCount - 1
            strList = Me.localiza.Column(1, I)
            If UCase$(Left(strList, intPos)) = strProcura Then
                Me.localiza.Selected(I) = True
                Selecionado = True
                Exit For
            Else
                Me.localiza.Selected(I) = False
                Selecionado = False
            End If
        Next
    End If
    
End Sub
